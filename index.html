Google Apps Script (GAS) 接收端設定與教學

提示：系統管理者亦可直接編輯本 HTML 檔案第 450 行的 DEFAULT_GAS_URL 常數直接寫死。

4 步驟快速建立接收端 Google 試算表：

  1. 開啟一張全新的 Google 試算表。
  2. 點選上方選單 「擴充功能」 > 「Apps Script」。
  3. 將下方的程式碼完全覆蓋貼入專案中，並點選右上方 「部署」 > 「新增部署」。
  4. 種類選擇 「網頁應用程式」，將執行身份設為「我」，誰有存取權限設為 「任何人 (Anyone)」，最後點擊部署並複製取得的 Web App URL。
Google Apps Script 接收端獨立程式碼： 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 建立主工作表
    var sheetName = (data.unitInfo.libName || "對照表寫入") + "_" + data.unitInfo.staffId;
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    } else {
      sheet.clear(); // 若重複送出則覆蓋
    }
    
    // 1. 寫入單位基本資訊
    sheet.appendRow(["【單位基本資訊】"]);
    sheet.appendRow(["圖書館名稱", data.unitInfo.libName]);
    sheet.appendRow(["圖書館代碼", data.unitInfo.libCode]);
    sheet.appendRow(["填表人代號", data.unitInfo.staffId]);
    sheet.appendRow(["電子郵件", data.unitInfo.staffEmail]);
    sheet.appendRow(["填表日期", data.unitInfo.fillDate]);
    sheet.appendRow([]);
    
    // 2. 寫入 942/952 館藏段對應
    sheet.appendRow(["【942 / 952 館藏段對應表】"]);
    sheet.appendRow(["標題 Tag/Subfield", "原系統對應段號", "實際資料範例", "備註"]);
    data.marcFields.forEach(function(row) {
      sheet.appendRow([row.label, row.source, row.sample, row.note]);
    });
    sheet.appendRow([]);
    
    // 3. 寫入各項對照代碼表
    function appendTable(title, headers, rows) {
      sheet.appendRow(["【" + title + "】"]);
      sheet.appendRow(headers);
      rows.forEach(function(r) {
        var rowArr = [];
        headers.forEach(function(h, idx) {
          rowArr.push(r["col" + idx] || "");
        });
        sheet.appendRow(rowArr);
      });
      sheet.appendRow([]);
    }
    
    appendTable("952$a/b 分館對照", ["圖書館名稱", "代碼"], data.tables.branches);
    appendTable("952$c 排架位置對照", ["排架位置", "代碼"], data.tables.locations);
    appendTable("942$c/952$y 館藏類型對照", ["館藏類型", "代碼", "可外借"], data.tables.itypes);
    appendTable("952$8 特藏號對照", ["特藏號名稱", "代碼"], data.tables.ccodes);
    appendTable("942$2/952$2 分類法對照", ["分類法名稱", "代碼"], data.tables.classSchemes);
    appendTable("讀者類型對照", ["讀者類型名稱", "代碼"], data.tables.patrons);
    appendTable("館藏狀態對照", ["中文描述", "代碼"], data.tables.statuses);
    appendTable("館方特有欄位", ["欄位名稱", "說明代碼"], data.tables.customs);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
